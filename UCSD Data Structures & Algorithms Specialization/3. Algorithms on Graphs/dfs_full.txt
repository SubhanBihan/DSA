struct Edge {
    int to;
    int w;
};

vector<vector<Edge>> tree;

struct Frame {
    int node;
    size_t next_child;
};

vector<int> path;        // nodes on root → current
vector<int> weights;    // edge weights along the path (parallel to path)
int dist = 0;           // total weight of current path
vector<Frame> stack;

int root = 0;

stack.push_back({root, 0});
path.push_back(root);

while (!stack.empty()) {
    auto& top = stack.back();
    int u = top.node;

    // Here you have:
    // path  = root → ... → u
    // dist  = sum of weights along that path

    process(u, path, dist);

    if (top.next_child < tree[u].size()) {
        auto [v, w] = tree[u][top.next_child++];

        // go down
        stack.push_back({v, 0});
        path.push_back(v);
        weights.push_back(w);
        dist += w;
    } else {
        // backtrack
        stack.pop_back();
        path.pop_back();

        if (!weights.empty()) {
            dist -= weights.back();
            weights.pop_back();
        }
    }
}
